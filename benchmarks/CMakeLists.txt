cmake_minimum_required(VERSION 3.22)
project(BridgeFlowBenchmark LANGUAGES C CXX)
set(LUA_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../build/tools/lua-5.4.8/src" CACHE PATH "Lua 5.4 source directory")
file(GLOB LUA_SOURCES CONFIGURE_DEPENDS "${LUA_SOURCE_DIR}/*.c")
list(FILTER LUA_SOURCES EXCLUDE REGEX "/lua(c)?[.]c$")
add_library(benchmark_lua STATIC ${LUA_SOURCES})
target_include_directories(benchmark_lua PUBLIC "${LUA_SOURCE_DIR}")
add_executable(BridgeFlowBenchmark FlowBenchmark.cpp)
target_compile_features(BridgeFlowBenchmark PRIVATE cxx_std_20)
target_include_directories(BridgeFlowBenchmark PRIVATE ../mod/include)
target_link_libraries(BridgeFlowBenchmark PRIVATE benchmark_lua)
if(MSVC)
    target_compile_options(BridgeFlowBenchmark PRIVATE /W4 /WX /permissive-)
endif()
